$ProjectPath = "C:\Users\Ishu\OneDrive\Documents\New project"
$ActivateVenv = ".\.venv\Scripts\Activate.ps1"

Write-Host "Starting General Research AI Agent..." -ForegroundColor Green

Set-Location $ProjectPath

if (!(Test-Path $ActivateVenv)) {
    Write-Host "Virtual environment not found (.venv missing)" -ForegroundColor Red
    exit
}

Write-Host "Starting Ollama..." -ForegroundColor Cyan
Start-Process powershell -ArgumentList @(
    "-NoExit",
    "-Command",
    "ollama serve"
)

Start-Sleep -Seconds 5

Write-Host "Starting Streamlit..." -ForegroundColor Cyan
Start-Process powershell -ArgumentList @(
    "-NoExit",
    "-Command",
    "cd '$ProjectPath'; & '$ProjectPath\.venv\Scripts\Activate.ps1'; streamlit run app.py"
)

Start-Sleep -Seconds 5

Start-Process "http://localhost:8501"

Write-Host ""
Write-Host "AI Agent Started Successfully" -ForegroundColor Green
Write-Host "Dashboard: http://localhost:8501" -ForegroundColor Yellow
